🚀 Flutter Bloc - API GET Request ရှင်းလင်းချက်

💡 စားသောက်ဆိုင် ဥပမာ (Restaurant Analogy)

API ကနေ Data ယူတာကို စားသောက်ဆိုင်မှာ "မီနူးစာအုပ် တောင်းကြည့်ခြင်း" နဲ့ ယှဉ်ကြည့်နိုင်ပါတယ် -

  • UI (Customer): စားသုံးသူက မီနူးစာအုပ် ကြည့်ချင်တယ်။
  • Event (FetchMenuEvent): စားပွဲထိုးကို "မီနူးစာအုပ် ခဏပေးပါ" လို့ တောင်းဆိုလိုက်တယ်။
  • Bloc (Chef/Waiter): စားပွဲထိုးက မီးဖိုချောင်/စတာရူးဆီ သွားပြီး မီနူးစာအုပ် သွားယူတယ်။
  • State (MenuState):
    • LoadingState: မီနူးစာအုပ် သွားယူနေတုန်း စောင့်ဆိုင်းခိုင်းထားခြင်း။
    • LoadedState: မီနူးစာအုပ် ရလာပြီး စားသုံးသူကို ပြသခြင်း။
    • ErrorState: မီနူးစာအုပ် ပျောက်နေ၍ စားသုံးသူကို အဆင်မပြေကြောင်း ပြောခြင်း။

💻 Complete Flutter Code Example

အောက်ပါ Code ကို ကော်ပီယူ၍ Flutter Project ၏ main.dart တွင် ထည့်သွင်း စမ်းသပ်နိုင်ပါသည်။ (http နှင့် flutter_bloc package ထည့်ရန် လိုအပ်ပါသည်)

main.dart 
import 'dart:convert';
import 'package:flutter/material.dart';
import 'package:flutter_bloc/flutter_bloc.dart';
import 'package:http/http.dart' as http;

// 1. MODEL
class Todo {
  final int id;
  final String title;

  Todo({required this.id, required this.title});

  factory Todo.fromJson(Map<String, dynamic> json) {
    return Todo(id: json['id'], title: json['title']);
  }
}

// 2. REPOSITORY (API GET)
class TodoRepository {
  Future<List<Todo>> fetchTodos() async {
    final response = await http.get(Uri.parse('https://jsonplaceholder.typicode.com/todos?_limit=5'));
    if (response.statusCode == 200) {
      final List data = json.decode(response.body);
      return data.map((e) => Todo.fromJson(e)).toList();
    } else {
      throw Exception('Failed to load data');
    }
  }
}

// 3. BLOC EVENTS & STATES
abstract class TodoEvent {}
class FetchTodosEvent extends TodoEvent {}

abstract class TodoState {}
class TodoLoadingState extends TodoState {}
class TodoLoadedState extends TodoState {
  final List<Todo> todos;
  TodoLoadedState(this.todos);
}
class TodoErrorState extends TodoState {
  final String message;
  TodoErrorState(this.message);
}

// 4. BLOC CLASS
class TodoBloc extends Bloc<TodoEvent, TodoState> {
  final TodoRepository repository;

  TodoBloc(this.repository) : super(TodoLoadingState()) {
    on<FetchTodosEvent>((event, emit) async {
      emit(TodoLoadingState());
      try {
        final todos = await repository.fetchTodos();
        emit(TodoLoadedState(todos));
      } catch (e) {
        emit(TodoErrorState(e.toString()));
      }
    });
  }
}

// 5. MAIN UI
void main() {
  runApp(
    RepositoryProvider(
      create: (context) => TodoRepository(),
      child: BlocProvider(
        create: (context) => TodoBloc(context.read<TodoRepository>())..add(FetchTodosEvent()),
        child: const MaterialApp(home: TodoListScreen()),
      ),
    ),
  );
}

class TodoListScreen extends StatelessWidget {
  const TodoListScreen({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('API GET via BLoC')),
      body: BlocBuilder<TodoBloc, TodoState>(
        builder: (context, state) {
          if (state is TodoLoadingState) {
            return const Center(child: CircularProgressIndicator());
          } else if (state is TodoLoadedState) {
            return ListView.builder(
              itemCount: state.todos.length,
              itemBuilder: (context, index) => ListTile(
                title: Text(state.todos[index].title),
              ),
            );
          } else if (state is TodoErrorState) {
            return Center(child: Text(state.message));
          }
          return const SizedBox.shrink();
        },
      ),
    );
  }
}
