// UI (Screen) Code
class ItemDetailScreen extends StatelessWidget {
  final int itemId;
  final TextEditingController _controller = TextEditingController();

  ItemDetailScreen({required this.itemId});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text('Update / Delete Item')),
      body: BlocConsumer<ActionBloc, ActionState>(
        listener: (context, state) {
          if (state is UpdateSuccessState) {
            ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Updated Successfully')));
          } else if (state is DeleteSuccessState) {
            ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Deleted Successfully')));
            Navigator.pop(context); // Go back after delete
          } else if (state is FailureState) {
            ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Error: ${state.error}')));
          }
        },
        builder: (context, state) {
          return Padding(
            padding: const EdgeInsets.all(16.0),
            child: Column(
              children: [
                TextField(
                  controller: _controller,
                  decoration: InputDecoration(labelText: 'Edit Data'),
                ),
                SizedBox(height: 20),
                Row(
                  mainAxisAlignment: MainAxisAlignment.spaceEvenly,
                  children: [
                    if (state is UpdatingState)
                      CircularProgressIndicator()
                    else
                      ElevatedButton(
                        onPressed: () {
                          final updatedData = DataModel(text: _controller.text);
                          context.read<ActionBloc>().add(UpdateDataEvent(itemId, updatedData));
                        },
                        child: Text('Update'),
                      ),
                    
                    if (state is DeletingState)
                      CircularProgressIndicator(color: Colors.red)
                    else
                      ElevatedButton(
                        style: ElevatedButton.styleFrom(backgroundColor: Colors.red),
                        onPressed: () {
                          context.read<ActionBloc>().add(DeleteDataEvent(itemId));
                        },
                        child: Text('Delete'),
                      ),
                  ],
                ),
              ],
            ),
          );
        },
      ),
    );
  }
}
// BLoC Logic Code
class ActionBloc extends Bloc<ActionEvent, ActionState> {
  final Repository repository;

  ActionBloc(this.repository) : super(ActionInitial()) {
    on<UpdateDataEvent>((event, emit) async {
      emit(UpdatingState());
      try {
        await repository.updateItem(event.id, event.data);
        emit(UpdateSuccessState());
      } catch (e) {
        emit(FailureState(e.toString()));
      }
    });

    on<DeleteDataEvent>((event, emit) async {
      emit(DeletingState());
      try {
        await repository.deleteItem(event.id);
        emit(DeleteSuccessState());
      } catch (e) {
        emit(FailureState(e.toString()));
      }
    });
  }
}
// State Code
abstract class ActionState extends Equatable {
  @override
  List<Object> get props => [];
}

class ActionInitial extends ActionState {}

class UpdatingState extends ActionState {}
class DeletingState extends ActionState {}

class UpdateSuccessState extends ActionState {}
class DeleteSuccessState extends ActionState {}

class FailureState extends ActionState {
  final String error;
  FailureState(this.error);
  
  @override
  List<Object> get props => [error];
}
// Event Code
abstract class ActionEvent extends Equatable {
  @override
  List<Object> get props => [];
}

class UpdateDataEvent extends ActionEvent {
  final int id;
  final DataModel data;
  
  UpdateDataEvent(this.id, this.data);

  @override
  List<Object> get props => [id, data];
}

class DeleteDataEvent extends ActionEvent {
  final int id;
  
  DeleteDataEvent(this.id);

  @override
  List<Object> get props => [id];
}
